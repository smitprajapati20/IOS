import SwiftUI

struct Fruit: Identifiable{
    let id = UUID()
    let name : String
    let color : String
    let price : Double
}

struct FruitList : View{
    let fruits = [
        Fruit(name: "Apple", color: "Red", price: 2.50),
        Fruit(name: "Banana", color: "Yellow", price: 1.20),
        Fruit(name: "Orange", color: "Orange", price: 3.00),
        Fruit(name: "Grapes", color: "Purple", price: 4.50),
        Fruit(name: "Mango", color: "Yellow", price: 5.00),
        Fruit(name: "Watermelon", color: "Green", price: 4.00),
        Fruit(name: "Kiwi", color: "Brown", price: 3.80),
        Fruit(name: "Strawberry", color: "Red", price: 3.50),
    ]

    var body: some View{
        NavigationStack {
            List(fruits) {
                fruit in NavigationLink(destination:FruitDetailView(fruit: fruit)){
                    HStack{
                        Text(fruit.name)
                        Spacer()
                        Text(String(format: "$%.2f", fruit.price))
                        .forgroundStyle(.secondary)
                    }
                }
            }
        }
    }
}

struct FruitDetailView: View{
    let fruit: Fruit
    var body: some View{
        VStack{
            Text(fruit.name)
            Text("Color: \(fruit.color)")
            Text(String(format: "Price : $%.2f", fruit.price))
        }
        .padding()
        .frame(maxWidth: .infinity, maxHeight: .infinity)
        .navigationTitle("Fruit Details")
        .navigationBarTitleDisplayMode(.inline)
    }
}
