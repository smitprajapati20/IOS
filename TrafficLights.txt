import SwiftUI

struct TrafficLight: View{
    @State private var light = 0
        var body: some View{
            VStack{
                Text("TrafficLight")

                Circle()
                .fill(Color.red)
                .frame(width:80,height:80)
                .opasity(light == 0 ? 1 : 0.2)'

                Circle()
                .fill(Color.yellow)
                .frame(width:80,height:80)
                .opasity(light == 2 ? 1 : 0.2)

                Circle()
                .fill(Color.green)
                .frame(width:80,height:80)
                .opasity(light == 1 ? 1 : 0.2)

                Text(light == 0 ? "STOP" : light == 1 ? "GO" : "WAIT")

                Button("Next"){
                    light = ( light + 1 ) % 3
                }
                .padding()
            }
        }
}

    TrafficLight()
}
